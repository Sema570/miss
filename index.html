# -*- coding: utf-8 -*- """ ================================================================================ ПРОЕКТ: Student Performance — Прогнозирование успеваемости студентов ВЕРСИЯ: 1.0 ДАТА: 01.04.2026 АВТОР: Data Science Team ================================================================================ Описание: Обучение модели регрессии, расчёт метрик ошибок (MSE), прогнозирование для новых студентов и генерация отчёта ================================================================================ """ import pandas as pd import numpy as np from sklearn.model_selection import train_test_split from sklearn.linear_model import LinearRegression from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score from sklearn.preprocessing import MinMaxScaler from datetime import datetime # ============================================================================== # 🔧 КОНФИГУРАЦИЯ # ============================================================================== RANDOM_STATE = 42 TEST_SIZE = 0.2 DATA_PATH = 'Downloads/StudentsPerformance.csv' # ============================================================================== # 1. ЗАГРУЗКА И ПОДГОТОВКА ДАННЫХ # ============================================================================== def load_and_prepare_data(path=DATA_PATH): """Загрузка данных и нормализация целевой переменной""" try: data = pd.read_csv(path) # Признаки и целевая переменная X = data[['math score']].values y = data['reading score'].values # Нормализация y к диапазону [0, 1] для малых значений MSE scaler = MinMaxScaler() y_scaled = scaler.fit_transform(y.reshape(-1, 1)).ravel() print(f"✅ Данные загружены: {len(data)} записей") return X, y_scaled, scaler, data except FileNotFoundError: print(f"❌ Файл {path} не найден!") return None, None, None, None # ============================================================================== # 2. ОБУЧЕНИЕ МОДЕЛИ # ============================================================================== def train_model(X, y): """Обучение линейной регрессии""" X_train, X_test, y_train, y_test = train_test_split( X, y, test_size=TEST_SIZE, random_state=RANDOM_STATE ) model = LinearRegression() model.fit(X_train, y_train) y_train_pred = model.predict(X_train) y_test_pred = model.predict(X_test) print(f"✅ Модель обучена (train: {len(X_train)}, test: {len(X_test)})") return model, X_train, X_test, y_train, y_test, y_train_pred, y_test_pred # ============================================================================== # 3. РАСЧЁТ МЕТРИК # ============================================================================== def calculate_metrics(y_train, y_train_pred, y_test, y_test_pred): """Расчёт всех метрик качества""" metrics = { 'mse_train': mean_squared_error(y_train, y_train_pred), 'mse_test': mean_squared_error(y_test, y_test_pred), 'rmse_train': np.sqrt(mean_squared_error(y_train, y_train_pred)), 'rmse_test': np.sqrt(mean_squared_error(y_test, y_test_pred)), 'mae_train': mean_absolute_error(y_train, y_train_pred), 'mae_test': mean_absolute_error(y_test, y_test_pred), 'r2_train': r2_score(y_train, y_train_pred), 'r2_test': r2_score(y_test, y_test_pred) } return metrics # ============================================================================== # 4. ПРОГНОЗ ДЛЯ НОВОГО СТУДЕНТА # ============================================================================== def predict_new_student(model, hours, scaler): """Прогноз баллов для одного студента""" # Создаём DataFrame как в примере new_student = pd.DataFrame({'math score': [hours * 10]}) # Масштабируем часы к оценкам prediction_scaled = model.predict(new_student)[0] prediction = prediction_scaled * 100 # Обратная нормализация return prediction # ============================================================================== # 5. ПРОГНОЗ ДЛЯ ГРУППЫ СТУДЕНТОВ # ============================================================================== def predict_group_students(model, hours_list, scaler): """Прогноз для группы студентов""" students = pd.DataFrame({'math score': [h * 10 for h in hours_list]}) predictions_scaled = model.predict(students) predictions = predictions_scaled * 100 # Обратная нормализация results = pd.DataFrame({ 'Часы подготовки': hours_list, 'Прогноз баллов': np.round(predictions, 2) }) return results # ============================================================================== # 6.
