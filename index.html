🛠 Исходный код модели

# -*- coding: utf-8 -*-
"""
Проект: Анализ успеваемости студентов (Student Performance Analysis)
Дата версии: 01.04.2026
Автор: Data Science Team
"""

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import MinMaxScaler

# Загрузка данных
data = pd.read_csv('Downloads/StudentsPerformance.csv')
X = data[['math score']].values
y = data['reading score'].values

# Нормализация
scaler = MinMaxScaler()
y_scaled = scaler.fit_transform(y.reshape(-1, 1)).ravel()

# Обучение
X_train, X_test, y_train, y_test = train_test_split(X, y_scaled, test_size=0.2, random_state=42)
model = LinearRegression()
model.fit(X_train, y_train)

# Оценка
mse_test = mean_squared_error(y_test, model.predict(X_test))
print(f"MSE: {mse_test:.4f}")
